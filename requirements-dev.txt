# Test-only dependencies. Herd's RUNTIME is stdlib-only: the plugin imports nothing
# outside the Python standard library in the gateway process, so an install never needs
# these. They exist to run the verification suite the README's Development block quotes.
#
#   python -m venv .venv && .venv/bin/pip install -r requirements-dev.txt
#   .venv/bin/python -m pytest tests/ -q
#
# Every one of these is optional to the suite: tests that need fastapi/httpx, PyYAML or
# aiohttp are skip-guarded, so a bare `pytest` run reports skips rather than errors.
# aiohttp runs the peer-route tests (tests/test_peer_secret_scope.py) and, with uvicorn,
# the two-gateway SPINE harness (scripts/qa/two_gateway_run.py).
pytest
fastapi
httpx
pyyaml
aiohttp
